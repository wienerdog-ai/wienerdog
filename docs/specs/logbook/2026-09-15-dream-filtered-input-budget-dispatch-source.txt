 assert.ok(kept.reduce((n,e)=>n+e.messages.length,0)<290);

test('A11: merged lock keeps live/unknown owner and recovers dead local owner', () => {
 const lock=require(path.join(root,'src/core/dream/lock'));
 const dir=fs.mkdtempSync(path.join(os.tmpdir(),'wd-content-lock-'));
 const file=path.join(dir,'dream.lock');
 try {
 const payload=JSON.stringify({pid:process.pid,host:os.hostname(),startedAt:new Date(0).toISOString(),deadline:0});
 fs.writeFileSync(file,payload); assert.equal(lock.acquireLock(dir,1000).reason,'busy'); assert.equal(fs.readFileSync(file,'utf8'),payload);
 fs.writeFileSync(file,'broken'); assert.equal(lock.acquireLock(dir,1000).reason,'owner-unknown'); assert.equal(fs.readFileSync(file,'utf8'),'broken');
 const child=cp.spawnSync(process.execPath,['-e','process.exit(0)']); assert.equal(child.status,0);
 fs.writeFileSync(file,JSON.stringify({pid:child.pid,host:os.hostname(),startedAt:new Date(0).toISOString(),deadline:0}));
 assert.deepEqual(lock.acquireLock(dir,1000),{acquired:true,stolen:true});
 } finally {fs.rmSync(dir,{recursive:true,force:true});}
});
